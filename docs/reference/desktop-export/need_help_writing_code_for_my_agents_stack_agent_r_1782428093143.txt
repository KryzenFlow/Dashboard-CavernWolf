2. docker-compose.yml
Yamlversion: "3.9"
services:
  reasoning_agent:
    build: ./reasoning_agent
    depends_on:
      - llama
    environment:
      - MEMORY_URL=http://memory_agent:5001
      - LLaMA_URL=http://llama:8000
    networks: [agents_net]

  memory_agent:
    build: ./memory_agent
    networks: [agents_net]

  agent_claw:
    build: ./agent_claw
    depends_on:
      - reasoning_agent
      - memory_agent
    networks: [agents_net]

  llama:
    build: ./llama_service
    ports:
      - "8000:8000"
    networks: [agents_net]

networks:
  agents_net:
    driver: bridge


3. Reasoning Agent (reasoning_agent/main.py)
Pythonfrom flask import Flask, request, jsonify
import requests

app = Flask(__name__)

@app.route("/reason", methods=["POST"])
def reason():
    data = request.json
    # Ask LLaMA for reasoning
    llama_resp = requests.post("http://llama:8000/generate", json={"prompt": data["query"]}).json()
    # Store in memory
    requests.post("http://memory_agent:5001/store", json={"query": data["query"], "response": llama_resp})
    return jsonify({"decision": llama_resp})

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)


4. Memory Agent (memory_agent/main.py)
Pythonfrom flask import Flask, request, jsonify

app = Flask(__name__)
memory_store = []

@app.route("/store", methods=["POST"])
def store():
    memory_store.append(request.json)
    return jsonify({"status": "stored"})

@app.route("/recall", methods=["GET"])
def recall():
    return jsonify(memory_store)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5001)


5. Agent Claw (Orchestrator) (agent_claw/main.py)
Pythonfrom flask import Flask, request, jsonify
import requests

app = Flask(__name__)

@app.route("/task", methods=["POST"])
def task():
    query = request.json["query"]
    decision = requests.post("http://reasoning_agent:5000/reason", json={"query": query}).json()
    # Here you could trigger website rebuild logic
    return jsonify({"result": decision})

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5002)

✅ How this works:

Agent Claw receives a job → sends it to Reasoning Agent
Reasoning Agent queries LLaMA → stores results in Memory Agent
Memory Agent keeps context for future tasks
