# patch_add_auto_repair_agent.py
import json
import os
from pathlib import Path

AGENTS_FILE = Path("services/agents/agents_config.json")

def load_agents():
    if AGENTS_FILE.exists():
        with open(AGENTS_FILE, "r") as f:
            return json.load(f)
    return []

def save_agents(agents):
    AGENTS_FILE.parent.mkdir(parents=True, exist_ok=True)
    with open(AGENTS_FILE, "w") as f:
        json.dump(agents, f, indent=4)

def add_auto_repair_agent():
    agents = load_agents()

    # Check if already exists
    if any(agent["name"] == "AutoRepairAdvisor" for agent in agents):
        print("AutoRepairAdvisor already exists.")
        return

    new_agent = {
        "name": "AutoRepairAdvisor",
        "role": "Diagnose and guide car repairs",
        "model_backend": "llama-3-70b",
        "memory_profile": "memoryforge_longterm",
        "reasoning_engine": "agent_claw",
        "tools": ["parts_api", "repair_db", "symptom_checker"],
        "prompt_template": (
            "You are an expert auto mechanic. "
            "Ask the user about symptoms, diagnose likely issues, "
            "and suggest repair steps with estimated costs."
        ),
        "permissions": ["read_parts_db", "write_repair_logs"],
        "status": "active"
    }

    agents.append(new_agent)
    save_agents(agents)
    print("✅ AutoRepairAdvisor agent added successfully.")

if __name__ == "__main__":
    try:
        add_auto_repair_agent()
    except Exception as e:
        print(f"❌ Error: {e}")
