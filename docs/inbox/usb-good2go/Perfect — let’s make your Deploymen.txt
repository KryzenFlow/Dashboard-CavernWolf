2. Updated DeploymentAgent
This version:

Reads the best version from Redis.
Deploys it via your deploy.sh script.
Runs a health check on the deployed site.
Reads the performance/stability scores from Redis (set by VersionEvaluator).
Sends a push notification to your phone.

Python# services/deploymentagent/main.py
import os
import redis
import subprocess
import requests
import time
import json

REDIS_HOST = os.getenv("REDIS_HOST", "redis")
REDIS_PORT = int(os.getenv("REDIS_PORT", 6379))
RESULT_KEY = "best_version"
SCORES_KEY = "version_scores"

# Pushover credentials (get from https://pushover.net/)
PUSHOVER_USER_KEY = os.getenv("PUSHOVER_USER_KEY")
PUSHOVER_API_TOKEN = os.getenv("PUSHOVER_API_TOKEN")

def send_push_notification(title, message):
    if not PUSHOVER_USER_KEY or not PUSHOVER_API_TOKEN:
        print("⚠️ Push notification skipped (missing credentials).")
        return
    try:
        resp = requests.post("https://api.pushover.net/1/messages.json", data={
            "token": PUSHOVER_API_TOKEN,
            "user": PUSHOVER_USER_KEY,
            "title": title,
            "message": message
        })
        if resp.status_code == 200:
            print("📲 Push notification sent.")
        else:
            print(f"⚠️ Push notification failed: {resp.text}")
    except Exception as e:
        print(f"❌ Push notification error: {e}")

def health_check(url):
    try:
        resp = requests.get(url, timeout=5)
        return resp.status_code == 200
    except:
        return False

def main():
    r = redis.Redis(host=REDIS_HOST, port=REDIS_PORT, decode_responses=True)
    best_version = r.get(RESULT_KEY)
    scores_data = r.get(SCORES_KEY)
    scores = json.loads(scores_data) if scores_data else {}

    if not best_version:
        print("❌ No best version found. Run VersionEvaluator first.")
        return

    print(f"🚀 Deploying best version: {best_version}")
    try:
        subprocess.run(["bash", "deploy.sh", best_version], check=True)
    except subprocess.CalledProcessError as e:
        print(f"❌ Deployment failed: {e}")
        send_push_notification("Deployment Failed", f"Version {best_version} failed to deploy.")
        return

    # Wait for service to start
    time.sleep(5)

    # Health check
    deployed_url = os.getenv("DEPLOYED_URL", "http://localhost:8080")
    healthy = health_check(deployed_url)

    if healthy:
        print(f"✅ Deployment successful: {deployed_url}")
        perf = scores.get(best_version, {}).get("performance_score", "N/A")
        stab = scores.get(best_version, {}).get("stability_score", "N/A")
        send_push_notification(
            "Deployment Successful",
            f"Version {best_version} deployed.\nPerformance: {perf}\nStability: {stab}"
        )
    else:
        print("❌ Deployment health check failed.")
        send_push_notification("Deployment Failed", f"Version {best_version} failed health check.")

if __name__ == "__main__":
    main()

4. How It Works in Your Orchestrator

VersionEvaluator runs → scores all versions → stores best version + scores in Redis.
DeploymentAgent runs → deploys best version → health check → push notification.
You get a real-time alert on your phone with:

Version name
Performance score
