1. Automatic Rollback System
We’ll extend the DeploymentAgent so if a deployment fails health checks, it:

Looks up the last known good version from Redis.
Deploys that version immediately.
Sends you a push notification:
"❌ Deployment failed — rolled back to version2 (last stable)."

Updated DeploymentAgent with Rollback
Python# services/deploymentagent/main.py
import os
import redis
import subprocess
import requests
import time
import json

REDIS_HOST = os.getenv("REDIS_HOST", "redis")
REDIS_PORT = int(os.getenv("REDIS_PORT", 6379))
RESULT_KEY = "best_version"
SCORES_KEY = "version_scores"
LAST_GOOD_KEY = "last_good_version"

PUSHOVER_USER_KEY = os.getenv("PUSHOVER_USER_KEY")
PUSHOVER_API_TOKEN = os.getenv("PUSHOVER_API_TOKEN")

def send_push_notification(title, message):
    if not PUSHOVER_USER_KEY or not PUSHOVER_API_TOKEN:
        print("⚠️ Push notification skipped (missing credentials).")
        return
    try:
        resp = requests.post("https://api.pushover.net/1/messages.json", data={
            "token": PUSHOVER_API_TOKEN,
            "user": PUSHOVER_USER_KEY,
            "title": title,
            "message": message
        })
        if resp.status_code == 200:
            print("📲 Push notification sent.")
        else:
            print(f"⚠️ Push notification failed: {resp.text}")
    except Exception as e:
        print(f"❌ Push notification error: {e}")

def health_check(url):
    try:
        resp = requests.get(url, timeout=5)
        return resp.status_code == 200
    except:
        return False

def deploy_version(version):
    try:
        subprocess.run(["bash", "deploy.sh", version], check=True)
        return True
    except subprocess.CalledProcessError as e:
        print(f"❌ Deployment failed: {e}")
        return False

def main():
    r = redis.Redis(host=REDIS_HOST, port=REDIS_PORT, decode_responses=True)
    best_version = r.get(RESULT_KEY)
    scores_data = r.get(SCORES_KEY)
    scores = json.loads(scores_data) if scores_data else {}

    if not best_version:
        print("❌ No best version found. Run VersionEvaluator first.")
        return

    print(f"🚀 Deploying best version: {best_version}")
    if not deploy_version(best_version):
        send_push_notification("Deployment Failed", f"Version {best_version} failed to deploy.")
        rollback_version = r.get(LAST_GOOD_KEY)
        if rollback_version:
            print(f"🔄 Rolling back to {rollback_version}")
            deploy_version(rollback_version)
            send_push_notification("Rollback Performed", f"Rolled back to {rollback_version} after failure.")
        return

    time.sleep(5)
    deployed_url = os.getenv("DEPLOYED_URL", "http://localhost:8080")
    if health_check(deployed_url):
        print(f"✅ Deployment successful: {deployed_url}")
        r.set(LAST_GOOD_KEY, best_version)
        perf = scores.get(best_version, {}).get("performance_score", "N/A")
        stab = scores.get(best_version, {}).get("stability_score", "N/A")
        send_push_notification(
            "Deployment Successful",
            f"Version {best_version} deployed.\nPerformance: {perf}\nStability: {stab}"
        )
    else:
        print("❌ Deployment health check failed.")
        send_push_notification("Deployment Failed", f"Version {best_version} failed health check.")
        rollback_version = r.get(LAST_GOOD_KEY)
        if rollback_version:
            print(f"🔄 Rolling back to {rollback_version}")
            deploy_version(rollback_version)
            send_push_notification("Rollback Performed", f"Rolled back to {rollback_version} after failure.")

if __name__ == "__main__":
    main()

Python# services/audiodiagnosis/main.py
import librosa
import numpy as np
import json
from sklearn.neighbors import KNeighborsClassifier
import os

MODEL_FILE = "engine_sound_model.pkl"

def diagnose_sound(file_path):
    y, sr = librosa.load(file_path)
    mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=40)
    features = np.mean(mfcc.T, axis=0).reshape(1, -1)

    import joblib
    model = joblib.load(MODEL_FILE)
    prediction = model.predict(features)[0]
    return prediction

if __name__ == "__main__":
    test_file = "uploads/car_sound.wav"
    if os.path.exists(test_file):
        result = diagnose_sound(test_file)
        print(json.dumps({"diagnosis": result}))
