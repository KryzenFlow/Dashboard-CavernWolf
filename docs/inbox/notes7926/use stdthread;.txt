use std::time::Duration;
use std::sync::{Arc, Mutex};

error_level: u32,
current_error: Arc<Mutex<u32>>,
}

fn new(error_level: u32) -> Self {
Monitor {
error_level,
current_error: Arc::new(Mutex::new(0)),
}
}

    }
}

}
