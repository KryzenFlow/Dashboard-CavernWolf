A. Version Evaluator & Deployment (with Rollback & Pushover Alerts)
# services/deploymentagent/main.py
import os, redis, subprocess, requests, time, json

REDIS_HOST = os.getenv("REDIS_HOST", "redis")
RESULT_KEY = "best_version"
LAST_GOOD_KEY = "last_good_version"
PUSHOVER_USER_KEY = os.getenv("PUSHOVER_USER_KEY")
PUSHOVER_API_TOKEN = os.getenv("PUSHOVER_API_TOKEN")

def send_push_notification(title, message):
    if PUSHOVER_USER_KEY and PUSHOVER_API_TOKEN:
        requests.post("https://api.pushover.net/1/messages.json", data={
            "token": PUSHOVER_API_TOKEN, "user": PUSHOVER_USER_KEY, "title": title, "message": message
        })

def health_check(url):
    try: return requests.get(url, timeout=5).status_code == 200
    except: return False

def deploy_version(version):
    try:
        subprocess.run(["bash", "deploy.sh", version], check=True)
        return True
    except subprocess.CalledProcessError: return False

def main():
    r = redis.Redis(host=REDIS_HOST, port=int(os.getenv("REDIS_PORT", 6379)), decode_responses=True)
    best_version = r.get(RESULT_KEY)
    if not best_version: return
    
    if not deploy_version(best_version):
        send_push_notification("Deployment Failed", f"Version {best_version} failed to deploy.")
        rollback = r.get(LAST_GOOD_KEY)
        if rollback:
            deploy_version(rollback)
            send_push_notification("Rollback Performed", f"Rolled back to {rollback}.")
        return

    time.sleep(5)
    if health_check(os.getenv("DEPLOYED_URL", "http://localhost:8080")):
        r.set(LAST_GOOD_KEY, best_version)
        send_push_notification("Deployment Successful", f"Version {best_version} deployed.")
    else:
        send_push_notification("Deployment Failed", f"Version {best_version} failed health check.")
        rollback = r.get(LAST_GOOD_KEY)
        if rollback:
            deploy_version(rollback)
            send_push_notification("Rollback Performed", f"Rolled back to {rollback}.")
B. LangGraph Supervisor + Meta/Growth Agent (Autonomous)
from typing import TypedDict, List, Dict
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.sqlite import SqliteSaver

class AgentState(TypedDict):
    user_goal: str
    plan: List[str]
    current_step: int
    research_notes: List[str]
    code_artifacts: Dict[str, str]
    learnings: List[str]
    next_run_time: str

def supervisor(state: AgentState) -> AgentState:
    if not state.get("plan"):
        state["plan"] = ["initial_research", "debug_repo", "deploy", "reflect_and_update"]
        state["current_step"] = 0
    step = state["plan"][state["current_step"]]
    routing = {"initial_research": "researcher", "debug_repo": "coder", "deploy": "deployer", "reflect_and_update": "meta_agent"}
    return {"next_step": routing.get(step, "meta_agent")}

def meta_agent(state: AgentState) -> AgentState:
    # Mutates patch: updates roadmap based on learnings
    import datetime
    state.setdefault("learnings", []).append("Refactor deployment script next time.")
    state["next_run_time"] = (datetime.datetime.utcnow() + datetime.timedelta(days=1)).isoformat()
    state["current_step"] += 1
    return state

# Graph wiring...
graph = StateGraph(AgentState)
graph.add_node("supervisor", supervisor)
graph.add_node("meta_agent", meta_agent)
# (Add researcher, coder, deployer nodes similarly)
graph.set_entry_point("supervisor")
graph.add_conditional_edges("supervisor", lambda s: s["next_step"], {"meta_agent": "meta_agent", "final": END})
app = graph.compile(checkpointer=SqliteSaver.from_file("agent_memory.db"))
