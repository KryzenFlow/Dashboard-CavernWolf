```rust
use crate::state::{AppState, Gateway, AuthMethod};
use uuid::Uuid;
use std::sync::Arc;

pub fn add_gateway(state: &Arc<AppState>, name: &str, auth: AuthMethod) -> Gateway {
    let gw = Gateway { id: Uuid::new_v4(), name: name.to_string(), auth, paired: false };
    state.gateways.write().insert(gw.id, gw.clone());
    gw
}

pub fn pair_via_code(state: &Arc<AppState>, id: Uuid, code: &str) -> bool {
    let mut gws = state.gateways.write();
    if let Some(g) = gws.get_mut(&id) {
        if let AuthMethod::PairingCode(c) = &g.auth {
            if c == code { g.paired = true; return true; }
        }
    }
    false
}
```

```rust
use crate::state::{AppState, Agent, AgentStatus};

pub fn register_agent(state: &Arc<AppState>, name: &str, gateway_id: Uuid) -> Option<Agent> {
    if !state.gateways.read().contains_key(&gateway_id) { return None; }
    let a = Agent { id: Uuid::new_v4(), name: name.into(), gateway_id, status: AgentStatus::Idle };
    state.agents.write().insert(a.id, a.clone());
    Some(a)
}

pub fn set_status(state: &Arc<AppState>, id: Uuid, s: AgentStatus) -> bool {
    if let Some(a) = state.agents.write().get_mut(&id) { a.status = s; true } else { false }
}
```

## `src/token_harness.rs`

```rust
use serde::{Serialize, Deserialize};

/// Multi-stage lifecycle switch for the cloud agent stack token harness.
#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq, Eq)]
#[serde(rename_all = "lowercase")]
pub enum LifecyclePhase {
    Boot, Mint, Provision, Active, Rotate, Revoke, Cold, Fault,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct HarnessState {
    pub phase: LifecyclePhase,
    pub auto_advance: bool,
    pub tokens_minted: u64,
    pub tokens_active: u64,
}

impl Default for HarnessState {
    fn default() -> Self { Self { phase: LifecyclePhase::Boot, auto_advance: false, tokens_minted: 0, tokens_active: 0 } }
}

impl HarnessState {
    pub fn advance(&mut self) -> LifecyclePhase {
        use LifecyclePhase::*;
        self.phase = match self.phase {
            Boot => Mint, Mint => Provision, Provision => Active,
            Active => Rotate, Rotate => Revoke, Revoke => Cold,
            Cold => Boot, Fault => Boot,
        };
        if self.phase == Mint { self.tokens_minted += 1; self.tokens_active += 1; }
        if self.phase == Revoke && self.tokens_active > 0 { self.tokens_active -= 1; }
        self.phase
    }
    pub fn set_phase(&mut self, p: LifecyclePhase) { self.phase = p; }
}
```

## `src/routes.rs`

```rust
use axum::{Router, routing::{get, post, put}, extract::{State, Path}, Json};
use std::sync::Arc;
use serde_json::{json, Value};
use crate::state::*;
use crate::token_harness::LifecyclePhase;

pub fn api_router(s: Arc<AppState>) -> Router {
    Router::new()
        .route("/gateways", get(list_gateways).post(add_gateway))
        .route("/gateways/:id/pair", post(pair_gateway))
        .route("/agents", get(list_agents).post(add_agent))
        .route("/agents/:id/status", put(set_agent_status))
        .route("/harness", get(get_harness).put(set_harness))
        .route("/harness/advance", post(advance_harness))
        .with_state(s)
}

async fn list_gateways(State(s): State<Arc<AppState>>) -> Json<Value> {
    Json(json!(s.gateways.read().values().cloned().collect::<Vec<_>>()))
}

async fn add_gateway(State(s): State<Arc<AppState>>, Json(b): Json<Value>) -> Json<Value> {
    let name = b["name"].as_str().unwrap_or("Production Server").to_string();
    let auth = match b["auth"].as_str().unwrap_or("token") {
        "password" => AuthMethod::Password,
        "pairing" => AuthMethod::PairingCode(b["code"].as_str().unwrap_or("0000").to_string()),
        _ => AuthMethod::Token,
    };
    Json(json!(crate::gateway::add_gateway(&s, &name, auth)))
}

async fn pair_gateway(State(s): State<Arc<AppState>>, Path(id): Path<uuid::Uuid>, Json(b): Json<Value>) -> Json<Value> {
    let code = b["code"].as_str().unwrap_or("");
    Json(json!({"paired": crate::gateway::pair_via_code(&s, id, code)}))
}

async fn list_agents(State(s): State<Arc<AppState>>) -> Json<Value> {
    Json(json!(s.agents.read().values().cloned().collect::<Vec<_>>()))
}

async fn add_agent(State(s): State<Arc<AppState>>, Json(b): Json<Value>) -> Json<Value> {
    let gw = uuid::Uuid::parse_str(b["gateway_id"].as_str().unwrap_or("")).unwrap_or_default();
    Json(json!(crate::agent::register_agent(&s, b["name"].as_str().unwrap_or("agent"), gw)))
}

async fn set_agent_status(State(s): State<Arc<AppState>>, Path(id): Path<uuid::Uuid>, Json(b): Json<Value>) -> Json<Value> {
    let st = match b["status"].as_str().unwrap_or("idle") {
        "running" => AgentStatus::Running, "paused" => AgentStatus::Paused,
        "stopped" => AgentStatus::Stopped, _ => AgentStatus::Idle,
    };
    Json(json!({"ok": crate::agent::set_status(&s, id, st)}))
}

async fn get_harness(State(s): State<Arc<AppState>>) -> Json<Value> {
    Json(json!(*s.harness.read()))
}

async fn set_harness(State(s): State<Arc<AppState>>, Json(b): Json<Value>) -> Json<Value> {
    let phase = serde_json::from_value::<LifecyclePhase>(b["phase"].clone()).unwrap_or(LifecyclePhase::Boot);
    let mut h = s.harness.write();
    h.set_phase(phase);
    if let Some(aa) = b["auto_advance"].as_bool() { h.auto_advance = aa; }
    Json(json!(*h))
}

async fn advance_harness(State(s): State<Arc<AppState>>) -> Json<Value> {
    let mut h = s.harness.write();
    let p = h.advance();
    Json(json!({"phase": p, "tokens_minted": h.tokens_minted, "tokens_active": h.tokens_active}))
}
```

## 2. `bot/src/main.rs` — Rule-Based Stateful Watcher

This is your **natural bot** — no AI, no agent loop. It follows hard rules, watches lifecycle tokens, kills corrupted agents, and alerts on rule violations.

```rust
use chrono::Utc;
use log::{info, warn, error};
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::sync::{Arc, Mutex};
use std::time::Duration;
use tokio::time::interval;

#[derive(Debug, Clone, Serialize, Deserialize)]
struct AgentState {
    agent_id: String,
    lifecycle_token: u64,
    max_lifecycle: u64,
    error_count: u32,
    error_threshold: u32,
    flags: Vec<String>,
    last_heartbeat: String,
    status: AgentStatus,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
enum AgentStatus {
    Healthy,
    Warning,
    Corrupt,
    Terminated,
}

#[derive(Debug, Clone)]
struct RuleViolation {
    agent_id: String,
    rule: String,
    severity: u32,
    timestamp: String,
}

struct PurpleBot {
    agents: Arc<Mutex<HashMap<String, AgentState>>>,
    error_threshold: u32,
    max_flags_before_terminate: u32,
    backend_url: String,
}

impl PurpleBot {
    fn new(backend_url: String) -> Self {
        Self {
            agents: Arc::new(Mutex::new(HashMap::new())),
            error_threshold: 5,
            max_flags_before_terminate: 3,
            backend_url,
        }
    }

    /// Hardcoded rules — no LLM, no inference. Pure rule logic.
    fn evaluate_rules(&self, agent: &AgentState) -> Vec<RuleViolation> {
        let mut violations = Vec::new();
        let now = Utc::now();

        // Rule 1: Lifecycle exceeded
        if agent.lifecycle_token >= agent.max_lifecycle {
            violations.push(RuleViolation {
                agent_id: agent.agent_id.clone(),
                rule: "LIFECYCLE_EXCEEDED".into(),
                severity: 10,
                timestamp: now.to_rfc3339(),
            });
        }

        // Rule 2: Error threshold reached
        if agent.error_count >= agent.error_threshold {
            violations.push(RuleViolation {
                agent_id: agent.agent_id.clone(),
                rule: "ERROR_THRESHOLD".into(),
                severity: 7,
                timestamp: now.to_rfc3339(),
            });
        }

        // Rule 3: Greed/coercion detection via flag patterns
        let greed_flags = agent.flags.iter().any(|f| {
            f.contains("resource_hog") || f.contains("manipulation")
                || f.contains("coercion") || f.contains("greed")
        });
        if greed_flags {
            violations.push(RuleViolation {
                agent_id: agent.agent_id.clone(),
                rule: "GREED_PATTERN_DETECTED".into(),
                severity: 9,
                timestamp: now.to_rfc3339(),
            });
        }

        // Rule 4: Stale heartbeat (>60s)
        if let Ok(last) = agent.last_heartbeat.parse::<chrono::DateTime<Utc>>() {
            if (now - last).num_seconds() > 60 {
                violations.push(RuleViolation {
                    agent_id: agent.agent_id.clone(),
                    rule: "STALE_HEARTBEAT".into(),
                    severity: 5,
                    timestamp: now.to_rfc3339(),
                });
            }
        }

        violations
    }

    /// Apply state transition based on violations — the bot's "judgment"
    fn apply_judgment(&self, agent_id: &str, violations: &[RuleViolation]) {
        let mut agents = self.agents.lock().unwrap();
        if let Some(agent) = agents.get_mut(agent_id) {
            let max_severity = violations.iter().map(|v| v.severity).max().unwrap_or(0);
            let greed_violation = violations.iter().any(|v| v.rule == "GREED_PATTERN_DETECTED");
            let lifecycle_violation = violations.iter().any(|v| v.rule == "LIFECYCLE_EXCEEDED");

            // Termination logic — bot kills lifecycle, not the agent itself
            if greed_violation || lifecycle_violation || max_severity >= 10 {
                agent.status = AgentStatus::Terminated;
                error!(
                    "[PURPLE-BOT] TERMINATING agent {} — rule: {}, severity: {}",
                    agent_id,
                    violations.iter().map(|v| v.rule.as_str()).collect::<Vec<_>>().join(","),
                    max_severity
                );
                self.alert(agent_id, "TERMINATED", &violations);
            } else if max_severity >= 7 {
                agent.status = AgentStatus::Corrupt;
                warn!("[PURPLE-BOT] Agent {} marked CORRUPT", agent_id);
                self.alert(agent_id, "CORRUPT", &violations);
            } else if max_severity >= 4 {
                agent.status = AgentStatus::Warning;
                warn!("[PURPLE-BOT] Agent {} WARNING", agent_id);
            }
        }
    }

    fn alert(&self, agent_id: &str, status: &str, violations: &[RuleViolation]) {
        // Log alert — extend with Slack/email/webhook
        let alert_json = serde_json::json!({
            "agent_id": agent_id,
            "status": status,
            "violations": violations,
            "timestamp": Utc::now().to_rfc3339(),
        });
        println!("[ALERT] {}", serde_json::to_string_pretty(&alert_json).unwrap());

        // Push to backend token store
        let url = format!("{}/api/alerts", self.backend_url);
        let client = reqwest::blocking::Client::new();
        let _ = client.post(&url).json(&alert_json).send();
    }

    /// Main watch loop — listens, evaluates, judges. No inference.
    fn watch_loop(&self) {
        let mut tick = interval(Duration::from_secs(5));
        loop {
            tick.tick().await;
            info!("[PURPLE-BOT] Sweep at {}", Utc::now().to_rfc3339());

            // Pull agent states from backend
            let url = format!("{}/api/agents", self.backend_url);
            match reqwest::get(&url).await {
                Ok(resp) => {
                    if let Ok(states) = resp.json::<Vec<AgentState>>().await {
                        let mut agents = self.agents.lock().unwrap();
                        for state in states {
                            agents.insert(state.agent_id.clone(), state);
                        }
                    }
                }
                Err(e) => warn!("[PURPLE-BOT] Backend unreachable: {}", e),
            }

            // Evaluate each agent against rules
            let agent_ids: Vec<String> = {
                let agents = self.agents.lock().unwrap();
                agents.keys().cloned().collect()
            };

            for id in agent_ids {
                let agent = {
                    let agents = self.agents.lock().unwrap();
                    agents.get(&id).cloned()
                };
                if let Some(agent) = agent {
                    if agent.status == AgentStatus::Terminated {
                        continue;
                    }
                    let violations = self.evaluate_rules(&agent);
                    if !violations.is_empty() {
                        self.apply_judgment(&id, &violations);
                    }
                }
            }
        }
    }
}

#[tokio::main]
async fn main() {
    env_logger::init();
    let backend_url = std::env::var("BACKEND_URL").unwrap_or_else(|_| "http://backend:8080".into());
    let bot = PurpleBot::new(backend_url);

    info!("[PURPLE-BOT] Natural bot online. Rule-based. Stateful. Watchful.");
    info!("[PURPLE-BOT] Rules: lifecycle, error_threshold, greed_pattern, heartbeat");

    bot.watch_loop().await;
}
```

```rust
use actix_cors::Cors;
use actix_web::{web, App, HttpResponse, HttpServer};
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::sync::{Arc, Mutex};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
struct SyntaxToken {
    id: String,
    agent_id: String,
    language: String,        // rust, python, js, ts, java
    token_type: String,      // bug_fix, conversion, snippet
    abbreviated: String,     // short form — lightweight
    original_ref: String,    // github repo ref
    timestamp: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct AlertRecord {
    id: String,
    agent_id: String,
    status: String,
    violations: serde_json::Value,
    timestamp: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
struct AgentState {
    agent_id: String,
    lifecycle_token: u64,
    max_lifecycle: u64,
    error_count: u32,
    error_threshold: u32,
    flags: Vec<String>,
    last_heartbeat: String,
    status: String,
}

struct AppState {
    tokens: Arc<Mutex<Vec<SyntaxToken>>>,
    alerts: Arc<Mutex<Vec<AlertRecord>>>,
    agents: Arc<Mutex<HashMap<String, AgentState>>>,
}

async fn get_tokens(state: web::Data<AppState>) -> HttpResponse {
    let tokens = state.tokens.lock().unwrap().clone();
    HttpResponse::Ok().json(tokens)
}

async fn post_token(state: web::Data<AppState>, token: web::Json<SyntaxToken>) -> HttpResponse {
    let mut t = token.into_inner();
    if t.id.is_empty() { t.id = Uuid::new_v4().to_string(); }
    if t.timestamp.is_empty() { t.timestamp = chrono::Utc::now().to_rfc3339(); }
    state.tokens.lock().unwrap().push(t.clone());
    HttpResponse::Created().json(t)
}

async fn get_alerts(state: web::Data<AppState>) -> HttpResponse {
    let alerts = state.alerts.lock().unwrap().clone();
    HttpResponse::Ok().json(alerts)
}

async fn post_alert(state: web::Data<AppState>, alert: web::Json<serde_json::Value>) -> HttpResponse {
    let record = AlertRecord {
        id: Uuid::new_v4().to_string(),
        agent_id: alert["agent_id"].as_str().unwrap_or("").to_string(),
        status: alert["status"].as_str().unwrap_or("").to_string(),
        violations: alert["violations"].clone(),
        timestamp: chrono::Utc::now().to_rfc3339(),
    };
    state.alerts.lock().unwrap().push(record.clone());
    HttpResponse::Created().json(record)
}

async fn get_agents(state: web::Data<AppState>) -> HttpResponse {
    let agents: Vec<AgentState> = state.agents.lock().unwrap().values().cloned().collect();
    HttpResponse::Ok().json(agents)
}

async fn post_agent(state: web::Data<AppState>, agent: web::Json<AgentState>) -> HttpResponse {
    state.agents.lock().unwrap().insert(agent.agent_id.clone(), agent.into_inner());
    HttpResponse::Ok().body("Registered")
}

async fn convert_code(payload: web::Json<ConvertRequest>) -> HttpResponse {
    // Lightweight syntax-token-based conversion stub
    // Extend with actual transpilation logic per language pair
    let req = payload.into_inner();
    let result = format!(
        "// Converted: {} -> {}\n// Token-abbreviated form:\n{}",
        req.from_lang, req.to_lang, req.code
    );
    HttpResponse::Ok().json(serde_json::json!({ "output": result }))
}

#[derive(Deserialize)]
struct ConvertRequest {
    from_lang: String,
    to_lang: String,
    code: String,
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    let state = web::Data::new(AppState {
        tokens: Arc::new(Mutex::new(Vec::new())),
        alerts: Arc::new(Mutex::new(Vec::new())),
        agents: Arc::new(Mutex::new(HashMap::new())),
    });

    println!("[BACKEND] Vellorae API on :8080");

    HttpServer::new(move || {
        let cors = Cors::permissive();
        App::new()
            .wrap(cors)
            .app_data(state.clone())
            .route("/api/tokens", web::get().to(get_tokens))
            .route("/api/tokens", web::post().to(post_token))
            .route("/api/alerts", web::get().to(get_alerts))
            .route("/api/alerts", web::post().to(post_alert))
            .route("/api/agents", web::get().to(get_agents))
            .route("/api/agents", web::post().to(post_agent))
            .route("/api/convert", web::post().to(convert_code))
    })
    .bind("0.0.0.0:8080")?
    .run()
    .await
}
```

## How the pieces connect

```
┌─────────────┐     tokens      ┌──────────────┐     alerts      ┌─────────────┐
│  Frontend   │ ──────────────> │   Backend    │ <────────────── │  Purple Bot │
│  Calculator │ <────────────── │  (Actix API) │ ──────────────> │  (Natural)  │
│  Token Zone │   GET /tokens   │  Token Store │  GET /agents    │  Rule-based │
└─────────────┘                 └──────────────┘                 └──────┬──────┘
                                                                        │
                                        evaluates rules every 5s ────────┘
                                        terminates on greed/lifecycle
                                        alerts on error threshold
```

**Key design decisions:**

1. **Natural bot** — pure rule logic, no LLM. Four hardcoded rules: lifecycle, error threshold, greed pattern, stale heartbeat. State transitions: `Healthy → Warning → Corrupt → Terminated`.

2. **Short lifecycle** — bot checks `lifecycle_token >= max_lifecycle` and terminates. You set `max_lifecycle` per agent on registration via `POST /api/agents`.

3. **Token drop zone** — agents push abbreviated syntax tokens to `POST /api/tokens`. Frontend polls every 10s. Each token is lightweight (abbreviated string + language + type + repo ref).

4. **Base64 toggle** — encoding layer on the calculator output for my encryption requirement.

5. **Save buttons** — each token has a SAVE button that downloads as JSON file to my local machine.

6. **Dockerized** — three containers: `backend`, `bot`, `frontend`. Bot has `restart: always` so it's always watching.

To deploy on vellorae.org via Cloudflare, put this behind a Cloudflare Tunnel or use Cloudflare Pages for the frontend + a VPS for backend+bot. The bot container is what makes it my purple team — it's the constant watcher in the middle

# services/deploymentagent/main.py
import os, redis, subprocess, requests, time, json

REDIS_HOST = os.getenv("REDIS_HOST", "redis")
RESULT_KEY = "best_version"
LAST_GOOD_KEY = "last_good_version"
PUSHOVER_USER_KEY = os.getenv("PUSHOVER_USER_KEY")
PUSHOVER_API_TOKEN = os.getenv("PUSHOVER_API_TOKEN")

def send_push_notification(title, message):
    if PUSHOVER_USER_KEY and PUSHOVER_API_TOKEN:
        requests.post("https://api.pushover.net/1/messages.json", data={
            "token": PUSHOVER_API_TOKEN, "user": PUSHOVER_USER_KEY, "title": title, "message": message
        })

def health_check(url):
    try: return requests.get(url, timeout=5).status_code == 200
    except: return False

def deploy_version(version):
    try:
        subprocess.run(["bash", "deploy.sh", version], check=True)
        return True
    except subprocess.CalledProcessError: return False

def main():
    r = redis.Redis(host=REDIS_HOST, port=int(os.getenv("REDIS_PORT", 6379)), decode_responses=True)
    best_version = r.get(RESULT_KEY)
    if not best_version: return
    
    if not deploy_version(best_version):
        send_push_notification("Deployment Failed", f"Version {best_version} failed to deploy.")
        rollback = r.get(LAST_GOOD_KEY)
        if rollback:
            deploy_version(rollback)
            send_push_notification("Rollback Performed", f"Rolled back to {rollback}.")
        return

    time.sleep(5)
    if health_check(os.getenv("DEPLOYED_URL", "http://localhost:8080")):
        r.set(LAST_GOOD_KEY, best_version)
        send_push_notification("Deployment Successful", f"Version {best_version} deployed.")
    else:
        send_push_notification("Deployment Failed", f"Version {best_version} failed health check.")
        rollback = r.get(LAST_GOOD_KEY)
        if rollback:
            deploy_version(rollback)
            send_push_notification("Rollback Performed", f"Rolled back to {rollback}.")
B. LangGraph Supervisor + Meta/Growth Agent (Autonomous)
from typing import TypedDict, List, Dict
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.sqlite import SqliteSaver

class AgentState(TypedDict):
    user_goal: str
    plan: List[str]
    current_step: int
    research_notes: List[str]
    code_artifacts: Dict[str, str]
    learnings: List[str]
    next_run_time: str

def supervisor(state: AgentState) -> AgentState:
    if not state.get("plan"):
        state["plan"] = ["initial_research", "debug_repo", "deploy", "reflect_and_update"]
        state["current_step"] = 0
    step = state["plan"][state["current_step"]]
    routing = {"initial_research": "researcher", "debug_repo": "coder", "deploy": "deployer", "reflect_and_update": "meta_agent"}
    return {"next_step": routing.get(step, "meta_agent")}

def meta_agent(state: AgentState) -> AgentState:
    # Mutates patch: updates roadmap based on learnings
    import datetime
    state.setdefault("learnings", []).append("Refactor deployment script next time.")
    state["next_run_time"] = (datetime.datetime.utcnow() + datetime.timedelta(days=1)).isoformat()
    state["current_step"] += 1
    return state

# Graph wiring...
graph = StateGraph(AgentState)
graph.add_node("supervisor", supervisor)
graph.add_node("meta_agent", meta_agent)
# (Add researcher, coder, deployer nodes similarly)
graph.set_entry_point("supervisor")
graph.add_conditional_edges("supervisor", lambda s: s["next_step"], {"meta_agent": "meta_agent", "final": END})
app = graph.compile(checkpointer=SqliteSaver.from_file("agent_memory.db"))
