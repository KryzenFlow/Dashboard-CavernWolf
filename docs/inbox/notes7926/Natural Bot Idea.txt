Example: Rust Supervisor Command to Kill Agent
Rustuse std::process::Command;

fn kill_agent(container_name: &str) -> Result<(), String> {
    let status = Command::new("docker")
        .args(&["stop", container_name])
        .status()
        .map_err(|e| e.to_string())?;
    if status.success() {
        Ok(())
    } else {
        Err(format!("Failed to stop container {}", container_name))
    }
}
