Key Safety Features
Short Life Cycles

Every agent container is spawned per request or per short session.
After task completion, container is destroyed.
No persistent memory unless explicitly allowed.
Token Harness

Supervisor tracks token usage for each agent and sub‑agent.
If usage exceeds quota → terminate container.
Prevents runaway loops or infinite API calls.
Deviation & Loop Detection

Supervisor monitors:
Execution time.
Repeated identical requests.
Unexpected output patterns.
If detected → kill container and log event.
Sub‑Agent Control

All sub‑agents must register with the parent agent.
Parent agent’s container is under Supervisor control.
Killing the parent automatically kills all sub‑agents.

use std::process::{Command, Stdio};
use std::time::{Duration, Instant};

struct AgentConfig {
    name: String,
    max_tokens: u32,
    max_runtime: Duration,
}

fn run_agent(config: AgentConfig, input: &str) -> Result<String, String> {
    let start_time = Instant::now();

    let mut child = Command::new("docker")
        .args(&["run", "--rm", &config.name])
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .spawn()
        .map_err(|e| e.to_string())?;

    if let Some(stdin) = child.stdin.as_mut() {
        stdin.write_all(input.as_bytes()).map_err(|e| e.to_string())?;
    }

    let output = child.wait_with_output().map_err(|e| e.to_string())?;

    // Runtime check
    if start_time.elapsed() > config.max_runtime {
        kill_agent(&config.name)?;
        return Err("Agent exceeded max runtime".into());
    }

    let stdout_str = String::from_utf8_lossy(&output.stdout).to_string();

    // Token check (simple example)
    let token_count = stdout_str.split_whitespace().count() as u32;
    if token_count > config.max_tokens {
        kill_agent(&config.name)?;
        return Err("Agent exceeded token limit".into());
    }

    Ok(stdout_str)
}

fn kill_agent(container_name: &str) -> Result<(), String> {
    let status = Command::new("docker")
        .args(&["kill", container_name])
        .status()
        .map_err(|e| e.to_string())?;
    if status.success() {
        Ok(())
    } else {
        Err(format!("Failed to kill container {}", container_name))
    }
}

5. Rust Supervisor (Tauri Backend)
src-tauri/src/main.rs
Rust#![cfg_attr(not(debug_assertions), windows_subsystem = "windows")]

use std::process::{Command, Stdio};
use std::io::Write;
use tauri::command;
use serde::{Serialize, Deserialize};
use std::time::{Duration, Instant};

#[derive(Serialize, Deserialize)]
struct ConvertRequest {
    code: String,
    from: String,
    to: String,
}

#[derive(Serialize, Deserialize)]
struct ConvertResponse {
    converted: String,
}

#[command]
fn run_langchange(req: ConvertRequest) -> Result<ConvertResponse, String> {
    let start_time = Instant::now();
    let max_runtime = Duration::from_secs(5);

    let mut child = Command::new("docker")
        .args(&["run", "--rm", "-i", "langchange_agent"])
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .spawn()
        .map_err(|e| e.to_string())?;

    if let Some(stdin) = child.stdin.as_mut() {
        let input_json = serde_json::to_string(&req).map_err(|e| e.to_string())?;
        stdin.write_all(input_json.as_bytes()).map_err(|e| e.to_string())?;
    }

    let output = child.wait_with_output().map_err(|e| e.to_string())?;

    if start_time.elapsed() > max_runtime {
        return Err("Agent exceeded max runtime".into());
    }

    let stdout_str = String::from_utf8_lossy(&output.stdout);
    serde_json::from_str::<ConvertResponse>(&stdout_str)
        .map_err(|e| format!("Parse error: {} | Raw: {}", e, stdout_str))
}

fn main() {
    tauri::Builder::default()
        .invoke_handler(tauri::generate_handler![run_langchange])
        .run(tauri::generate_context!())
        .expect("error while running tauri app");
}
