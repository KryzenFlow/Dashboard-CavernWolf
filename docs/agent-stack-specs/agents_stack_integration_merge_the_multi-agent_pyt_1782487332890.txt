1️⃣ Secure Command Whitelist for Hermes Studio
We’ll make sure /cli/run only allows safe commands that customers can use — no raw shell access.
hermes-server/routes/cli.js (updated)
Javascriptconst express = require("express");
const { spawn } = require("child_process");

const router = express.Router();

// Define allowed commands and args
const ALLOWED_COMMANDS = {
  "new": ["site", "app"],
  "deploy": ["github", "docker"],
  "ai": ["suggest-template", "generate-content"]
};

router.post("/run", (req, res) => {
  const { command, args } = req.body;

  if (!command || !ALLOWED_COMMANDS[command]) {
    return res.status(400).json({ error: "Command not allowed" });
  }

  // Validate subcommand
  if (args.length === 0 || !ALLOWED_COMMANDS[command].includes(args[0])) {
    return res.status(400).json({ error: "Subcommand not allowed" });
  }

  // Run hermes-cli inside dev-tools container
  const dockerCmd = ["exec", "-i", "dev-tools", "hermes-cli", command, ...args];
  const proc = spawn("docker", dockerCmd);

  res.setHeader("Content-Type", "text/plain");

  proc.stdout.on("data", (data) => res.write(data));
  proc.stderr.on("data", (data) => res.write(`ERROR: ${data}`));
  proc.on("close", (code) => {
    res.write(`\nProcess exited with code ${code}`);
    res.end();
  });
});

module.exports = router;

Jsximport React, { useState } from "react";

export default function WorkflowButtons() {
  const [output, setOutput] = useState("");

  const runCommand = async (command, args) => {
    setOutput(`Running: hermes-cli ${command} ${args.join(" ")}\n`);
    const res = await fetch("/cli/run", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ command, args })
    });

    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let done = false;

    while (!done) {
      const { value, done: readerDone } = await reader.read();
      done = readerDone;
      if (value) {
        setOutput((prev) => prev + decoder.decode(value));
      }
    }
  };

  return (
    <div>
      <h3>Quick Actions</h3>
      <button onClick={() => runCommand("new", ["site", "--template", "portfolio", "--name", "mysite"])}>
        New Portfolio Site
      </button>
      <button onClick={() => runCommand("deploy", ["github", "--repo", "https://<token>@github.com/user/mysite.git"])}>
        Deploy to GitHub Pages
      </button>
      <button onClick={() => runCommand("ai", ["suggest-template", "--industry", "real estate"])}>
        AI Suggest Template
      </button>
      <button onClick={() => runCommand("ai", ["generate-content", "--topic", "solar panels"])}>
        AI Generate Content
      </button>
      <pre style={{ background: "#111", color: "#0f0", padding: "10px" }}>
        {output}
      </pre>
    </div>
  );
}
