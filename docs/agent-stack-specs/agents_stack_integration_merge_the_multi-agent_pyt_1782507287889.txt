from agent_registry import load_registry, save_registry, hot_reload_agent, AgentConfig, active_agents

app = FastAPI(title="Agent Control API", version="2.0")

# ---------------------------
# REST Endpoints
# ---------------------------
@app.get("/agents")
def list_agents():
    return load_registry()

@app.post("/agents")
def create_agent(config: AgentConfig):
    registry = load_registry()
    if config.name in registry:
        raise HTTPException(status_code=400, detail="Agent already exists")
    registry[config.name] = config.dict()
    save_registry(registry)
    hot_reload_agent(config.name, config)
    return {"status": "created", "agent": config.name}

@app.put("/agents/{name}")
def update_agent(name: str, config: AgentConfig):
    registry = load_registry()
    if name not in registry:
        raise HTTPException(status_code=404, detail="Agent not found")
    registry[name] = config.dict()
    save_registry(registry)
    hot_reload_agent(name, config)
    return {"status": "updated", "agent": name}

@app.delete("/agents/{name}")
def delete_agent(name: str):
    registry = load_registry()
    if name not in registry:
        raise HTTPException(status_code=404, detail="Agent not found")
    del registry[name]
    save_registry(registry)
    active_agents.pop(name, None)
    return {"status": "deleted", "agent": name}

# ---------------------------
# WebSocket: Chat Streaming
# ---------------------------
@app.websocket("/ws/chat/{agent_name}")
async def chat_ws(websocket: WebSocket, agent_name: str):
    await websocket.accept()
    if agent_name not in active_agents:
        await websocket.send_text(json.dumps({"error": "Agent not found"}))
        await websocket.close()
        return
    try:
        while True:
            data = await websocket.receive_text()
            async for chunk in stream_agent_response(agent_name, json.loads(data)):
                await websocket.send_text(json.dumps({"chunk": chunk}))
    except WebSocketDisconnect:
        pass

# ---------------------------
# WebSocket: Log Streaming
# ---------------------------
@app.websocket("/ws/logs/{service_name}")
async def logs_ws(websocket: WebSocket, service_name: str):
    await websocket.accept()
    try:
        async for line in stream_logs(service_name):
            await websocket.send_text(line)

        pass

sessions: Dict[str, TerminalSession] = {}

@app.websocket("/ws/terminal/{session_id}")
async def terminal_ws(websocket: WebSocket, session_id: str):
    await websocket.accept()
    if session_id not in sessions:
        sessions[session_id] = TerminalSession()
    term = sessions[session_id]
    try:
        async for message in term.handle_io(websocket):
            pass
    except WebSocketDisconnect:
        term.close()
        del sessions[session_id]


chat_streamer.py — Streaming Agent Responses
Pythonimport asyncio
from agent_registry import active_agents

async def stream_agent_response(agent_name: str, payload: dict):
    """Yield chunks of agent output for streaming to UI."""
    agent = active_agents[agent_name]
    # Simulate streaming by splitting into words
    result = agent.run(payload)
    if isinstance(result, dict) and "output" in result:
        text = result["output"]
    else:
        text = str(result)
    for word in text.split():
        await asyncio.sleep(0.05)  # simulate token delay
        yield word


log_streamer.py — Docker Log Streaming
Pythonimport asyncio
import docker

client = docker.from_env()

async def stream_logs(service_name: str):
    """Stream logs from a running Docker service."""
    container = None
    for c in client.containers.list():
        if service_name in c.name:
            container = c
            break
    if not container:
        yield f"[ERROR] No container found for {service_name}"
        return
    for line in container.logs(stream=True, follow=True):
        yield line.decode("utf-8").rstrip()
        await asyncio.sleep(0)  # yield control


terminal_manager.py — Interactive PTY for xterm.js
Pythonimport asyncio
import os
import pty
import select
import subprocess

class TerminalSession:
    def __init__(self, shell="/bin/bash"):
        self.master_fd, self.slave_fd = pty.openpty()
        self.process = subprocess.Popen(
            [shell],
            preexec_fn=os.setsid,
            stdin=self.slave_fd,
            stdout=self.slave_fd,
            stderr=self.slave_fd,
            universal_newlines=True
        )

    async def handle_io(self, websocket):
        loop = asyncio.get_event_loop()
        while True:
            r, _, _ = select.select([self.master_fd], [], [], 0)
            if r:
                output = os.read(self.master_fd, 1024).decode()
                await websocket.send_text(output)
            try:
                data = await asyncio.wait_for(websocket.receive_text(), timeout=0.05)
                os.write(self.master_fd, data.encode())
            except asyncio.TimeoutError:
                pass

    def close(self):
        self.process.terminate()
        os.close(self.master_fd)
        os.close(self.slave_fd)
