Python#!/usr/bin/env python3
import click
import requests
import os

CLAW_URL = os.getenv("CLAW_URL", "http://agent-claw:9000")

@click.group()
def cli():
    """Minimal Hermes CLI"""
    pass

@cli.command()
@click.option("--template", default="default", help="Template name")
@click.option("--name", required=True, help="Project name")
def new_site(template, name):
    """Create a new site from a template."""
    payload = {
        "action": "build_website",
        "params": {"template": template, "output_dir": f"/shared/{name}"}
    }
    r = requests.post(f"{CLAW_URL}/execute", json=payload)
    click.echo(r.json())

if __name__ == "__main__":
    cli()

Express /cli/run Endpoint with Allowlist
routes/cli.js (new file)
Javascriptconst express = require("express");
const { spawn } = require("child_process");

const router = express.Router();

// Allowed commands
const ALLOWED = {
  "new_site": true
};

router.post("/run", (req, res) => {
  const { command, args = [] } = req.body;

  if (!ALLOWED[command]) {
    return res.status(400).json({ error: "Command not allowed" });
  }

  const proc = spawn("docker", ["exec", "-i", "dev-tools", "hermes-cli", command, ...args]);

  res.setHeader("Content-Type", "text/plain");

  proc.stdout.on("data", (data) => res.write(data));
  proc.stderr.on("data", (data) => res.write(`ERROR: ${data}`));
  proc.on("close", (code) => {
    res.write(`\nExited with code ${code}`);
    res.end();
  });
});

module.exports = router;
